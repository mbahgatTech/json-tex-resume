    % Plain-text contact details (no icon glyphs) so ATS parsers read them. Personal details (city, phone,
    % email) get their own line above the links, and \mbox keeps each item from breaking across lines.
    % A line too wide for the page wraps at a separator instead, and the separator is dropped there.
    \def\contactbar{\discretionary{}{}{\enspace\textbar\enspace}}
    \def\contactsep{}
    \ifx\candidateAddress\empty\else\contactsep\mbox{\candidateAddress}\let\contactsep\contactbar\fi
    \ifx\candidatePhone\empty\else\contactsep\mbox{\candidatePhone}\let\contactsep\contactbar\fi
    \ifx\candidateEmail\empty\else\contactsep\href{mailto:\candidateEmail}{\underline{\candidateEmail}}\let\contactsep\contactbar\fi
    \ifx\contactsep\empty\else \\ \vspace{1pt} \fi
    \def\contactsep{}
    \ifx\linkedin\empty\else\contactsep\href{https://linkedin.com/in/\linkedin/}{\underline{linkedin.com/in/\linkedin}}\let\contactsep\contactbar\fi
    \ifx\github\empty\else\contactsep\href{https://github.com/\github}{\underline{github.com/\github}}\let\contactsep\contactbar\fi
    \ifx\portfolio\empty\else\contactsep\href{https://www.\portfolio}{\underline{\portfolio}}\fi
